def attention(query, keys, values):
    # 1. Compute similarity scores
    scores = dot(query, keys.T)
    
    # 2. Softmax to get weights
    weights = softmax(scores)
    
    # 3. Weighted sum of values
    output = dot(weights, values)
    
    return output
